#+TITLE: linux memory reclaim

Linux 中内存分为 inactive 和 active 两个 LRU 链表，匿名内存和页缓存各有两个，inactive_anon、active_anon、inactive_file、active_file 组成 lruvec 向量。

内存回收时会根据内存大小确定 inactive 比例，同时尝试将 active 链上不活跃的部分移到 inactive 链，并对 inactive 链进行回收。

而 Working Set Size(WSS) 则是进程内存使用减去 inactive_file 的部分，内核 OOM 基于 WSS 部分考虑。WSS 可能随内存回收减小，但有可能有较多 active 内存无法回收，最后 OOM。

#+begin_src c
  static unsigned long shrink_list(enum lru_list lru, unsigned long nr_to_scan,
           struct lruvec *lruvec, struct scan_control *sc)
  {
    if (is_active_lru(lru)) {
      if (sc->may_deactivate & (1 << is_file_lru(lru)))
        shrink_active_list(nr_to_scan, lruvec, sc, lru);
      else
        sc->skipped_deactivate = 1;
      return 0;
    }
  
    return shrink_inactive_list(nr_to_scan, lruvec, sc, lru);
  }
#+end_src
